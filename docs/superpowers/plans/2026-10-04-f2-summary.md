**Goal:** 列上声明 `sum` 后,库自己合成合计行:远程 / 静态分页时显示「本页合计」+「全部合计」两行(窄档是两张合计卡),后端经 `PageResult.summary` 提供全量合计。

## 已定规则(来自设计 §13,测试要逐条覆盖)

| 场景 | 显示 |
|---|---|
| `pagination === false`(不分页) | 一行,标签「合计」(`labels.summary`) |
| 远程 + 分页 + 后端返回了 `summary` | 两行:「本页合计」(前端求和)+「全部合计」(后端值) |
| 远程 + 分页 + 后端没返回 `summary` | 一行「本页合计」 |
| 静态 + 分页 | 两行:「本页合计」+「全部合计」(过滤后的全部行求和) |
| 可编辑表格 | 两个口径都**不计待删行**、计入草稿值;「全部合计」的后端值不含草稿 |
| 求值 | `null` / `undefined` / `''` / 纯空白 / 非数字串 / `NaN` / `±Infinity` 一律按 0;`'12.5'` 按数字;不出现 `NaN` |
| 格式 | `format: 'money'` → 千分位两位小数;否则有 `dec` → `toFixed(dec)`;否则千分位整数(`toLocaleString('en-US')`) |
| 后端 `summary` 缺某列 / 值非有限数 | 那一格不画(不显示 0,免得误导) |

## 已知限制(写进设计与 README,不在本计划里解决)

- 标签落在「声明顺序里第一个可见、非合计、非特殊的叶子列」;用户在列设置里把这一列隐藏后标签随之不见(原型模块 5 的注释里早有同样的说明)。窄档卡片标签落在卡片标题列。
- 函数形式的 `format` 不参与合计格式化(没有合法的 `row` 实参可传),合计格按 `dec` / 千分位整数显示。

```ts
// 合计的纯函数(设计文档 13),与 Vue / Naive 无关,可单测(同 filter.ts 的定位)。
// SmartTable.vue 只负责接线:把这里合成的行交给官方 NDataTable 的 summary,窄档卡片读同一份。
import { formatMoney } from './format'
import type { SmartTableColumn, SmartTableDataColumn } from './types'
import { isSpecialColumn } from './useColumns'

/** 参与合计的叶子列。 */
export interface SumColumn {
  key: string
  /** 小数位;缺省千分位整数。 */
  dec?: number
  /** 列的 format 是 'money':千分位两位小数,优先于 dec。 */
  money: boolean
}

/** 一行合计:标签 + 各列的数值。 */
export interface SummaryLine {
  label: string
  values: Record<string, number>
}

/** 官方 CreateSummary 的行形状(每格 { value });库只用到 value。 */
export type SummaryCells = Record<string, { value: string }>

export interface SummaryLabels {
  summary: string
  summaryPage: string
  summaryTotal: string
}

export interface SummaryLinesInput {
  cols: readonly SumColumn[]
  labels: SummaryLabels
  /** props.pagination !== false。 */
  paginated: boolean
  isRemote: boolean
  /** 后端随分页结果返回的全量合计;没返回 / 静态模式为 null。 */
  remoteTotals: Record<string, number> | null
  /** 当前页的行(调用方已剔除待删行)。 */
  pageRows: ReadonlyArray<Record<string, unknown>>
  /** 静态模式的「全部行」(过滤后、已剔除待删行);惰性求值,用不到就不算。 */
  allRows: () => ReadonlyArray<Record<string, unknown>>
}

/** 全部叶子数据列(多级表头展开,跳过勾选 / 展开 / 序号等特殊列)。 */
function dataLeaves<T>(columns: ReadonlyArray<SmartTableColumn<T>>): SmartTableDataColumn<T>[] {
  const out: SmartTableDataColumn<T>[] = []
  const walk = (list: ReadonlyArray<SmartTableColumn<T>>): void => {
    for (const c of list) {
      if (isSpecialColumn(c)) continue
      if (c.children?.length) walk(c.children)
      else out.push(c)
    }
  }
  walk(columns)
  return out
}

export function sumColumns<T>(columns: ReadonlyArray<SmartTableColumn<T>>): SumColumn[] {
  return dataLeaves(columns)
    .filter((c) => c.sum)
    .map((c) => ({
      key: c.key,
      ...(c.dec !== undefined ? { dec: c.dec } : {}),
      money: c.format === 'money',
    }))
}

/**
 * 标签(「合计」「本页合计」…)放哪一列:声明顺序里第一个可见、非合计的叶子列;都不满足退回第一个叶子列。
 * 局限:用户在列设置里把这一列隐藏后标签随之不见(库的合计行不跟随列设置里的「第一个可见列」)。
 */
export function labelColumnKey<T>(
  columns: ReadonlyArray<SmartTableColumn<T>>,
  cols: readonly SumColumn[],
): string | undefined {
  const sums = new Set(cols.map((c) => c.key))
  const leaves = dataLeaves(columns)
  const visible = leaves.filter((c) => !c.hide && !c.hideInTable)
  return (visible.find((c) => !sums.has(c.key)) ?? leaves[0])?.key
}

/** 数字或数字串按数字;其余(含 NaN / ±Infinity / 空串 / 纯空白 / 非数字串)按 0,免得合计显示 NaN。 */
function toNumber(v: unknown): number {
  if (typeof v === 'number') return Number.isFinite(v) ? v : 0
  if (typeof v === 'string' && v.trim() !== '') {
    const n = Number(v)
    return Number.isFinite(n) ? n : 0
  }
  return 0
}

export function sumValues(
  rows: ReadonlyArray<Record<string, unknown>>,
  cols: readonly SumColumn[],
): Record<string, number> {
  const out: Record<string, number> = {}
  for (const c of cols) out[c.key] = rows.reduce((n, r) => n + toNumber(r[c.key]), 0)
  return out
}

/** 函数形式的 format 不参与:它需要一个合法的 row 实参,合计行没有。 */
export function formatSum(col: SumColumn, value: number): string {
  if (col.money) return formatMoney(value)
  if (col.dec !== undefined) return value.toFixed(col.dec)
  return value.toLocaleString('en-US')
}

/** 设计 §13 的显示规则:几行、各叫什么、数从哪来。 */
export function buildSummaryLines(i: SummaryLinesInput): SummaryLine[] {
  const page = sumValues(i.pageRows, i.cols)
  if (!i.paginated) return [{ label: i.labels.summary, values: page }]
  const pageLine: SummaryLine = { label: i.labels.summaryPage, values: page }
  if (i.isRemote) {
    return i.remoteTotals
      ? [pageLine, { label: i.labels.summaryTotal, values: i.remoteTotals }]
      : [pageLine]
  }
  return [pageLine, { label: i.labels.summaryTotal, values: sumValues(i.allRows(), i.cols) }]
}

export function buildSummaryRows(
  lines: readonly SummaryLine[],
  cols: readonly SumColumn[],
  labelKey: string | undefined,
): SummaryCells[] {
  return lines.map((line) => {
    const row: SummaryCells = {}
    if (labelKey !== undefined) row[labelKey] = { value: line.label }
    for (const c of cols) {
      const v = line.values[c.key]
      // 后端没给这一列(或给了非有限数):不画,不显示 0
      if (v === undefined || !Number.isFinite(v)) continue
      row[c.key] = { value: formatSum(c, v) }
    }
    return row
  })
}
```
